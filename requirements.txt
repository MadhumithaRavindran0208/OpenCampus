Flask==3.1.3
Werkzeug==3.1.7
Jinja2==3.1.6
itsdangerous==2.2.0
Markdown==3.10.2

# Optional but recommended: a battle-tested Rust-based HTML sanitizer.
# app/core/sanitize.py uses this automatically when it's installed, and
# falls back to a hand-rolled (but tested -- see tests/test_sanitize.py)
# stdlib sanitizer when it's not, so nothing breaks either way.

waitress==3.0.2
